async function abrirReproductor(item) { itemActivo = item; servidorActivoIdx = 0; temporadaActiva = 1; episodioActivo = 1; document.getElementById('playerModal').style.display = 'block'; servidoresDisponibles = []; try { // Consultar la BD local const resBackend = await fetch(`/api/contenido/${item.id}`); // VALIDACIÓN CLAVE: comprobar que la respuesta HTTP sea 200 OK y no 404 if (resBackend.ok) { const dataBackend = await resBackend.json(); if (dataBackend.success && dataBackend.data) { servidoresDisponibles.push({ nombre: "Servidor VIP (Sin Publicidad)", isBackend: true, backendId: dataBackend.data.id }); } } else { console.log(`El ID ${item.id} no está en la BD local. Cargando servidores externos.`); } } catch (e) { console.log("No se pudo conectar con la BD local, usando respaldos remotos."); } // Servidores de respaldo TMDB SERVIDORES_TMDB.forEach(srv => { servidoresDisponibles.push({ ...srv, isBackend: false }); }); renderizarBotonesServidores(); if (modoActual === 'movie') { document.getElementById('seriesPanel').style.display = 'none'; actualizarIframe(); } else { document.getElementById('seriesPanel').style.display = 'block'; document.getElementById('seriesTitle').innerText = item.name || "Serie"; await cargarEstructuraSerie(item.id); } }
